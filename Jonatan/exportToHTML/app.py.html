@app.route("/wordle")
def wordle():
    return render_template("wordle.html")


@app.route("/wordle/get_value")
def wordle_api():
    tunnus = random.randint(1, 1420)
    try:
        yhteys = get_connection()
        cursor = yhteys.cursor()
        query = "select sana from wordle where id = %s"
        cursor.execute(query, (tunnus,))
        result = cursor.fetchone()

    except mysql.connector.Error as err:
        print("Virhe: " , err)
    finally:
        if cursor:
            cursor.close()
        if yhteys:
            yhteys.close()

    if result:
        return jsonify({"value": result})
    return None


@app.route("/wordle/set_value", methods=['POST'])
def set_value_wordle():
    global wordle_win
    data = request.get_json()
    if not data or 'value' not in data:
        return jsonify({"error": "No value provided"}), 400

    wordle_win = bool(data['value'])
    print("wordle_win = ", wordle_win)
    return jsonify({"status": 'ok', 'received': wordle_win})
