from PIL import Image, ImageDraw

def make_circle(filename, size=256, fill=(160,160,160,255), outline=(130,130,130,255), outline_width=6):
    """ 
    Harmaa ympyrä: 
    - fill = keskiosa (tummempi) 
    - outline = reuna (vain vähän tummempi) 
    """
    img = Image.new("RGBA", (size, size), (0,0,0,0))
    draw = ImageDraw.Draw(img)
    draw.ellipse(
        [outline_width//2, outline_width//2, size-outline_width//2, size-outline_width//2],
        fill=fill,
        outline=outline,
        width=outline_width
    )
    img.save(filename)
    print("Saved:", filename)

def make_square(filename, size=256, fill=(160,160,160,255), outline=(130,130,130,255), outline_width=6, radius=30):
    """ 
    Harmaa neliö: 
    - fill = keskiosa (tummempi) 
    - outline = reuna (vain vähän tummempi) 
    """
    img = Image.new("RGBA", (size, size), (0,0,0,0))
    draw = ImageDraw.Draw(img)
    draw.rounded_rectangle(
        [outline_width//2, outline_width//2, size-outline_width//2, size-outline_width//2],
        radius=radius,
        fill=fill,
        outline=outline,
        width=outline_width
    )
    img.save(filename)
    print("Saved:", filename)

# luodaan kuvat
make_circle("static/img/greyCircle.png")
make_square("static/img/greySquare.png")
