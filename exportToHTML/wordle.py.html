def vastaus_wordle_def():
    """Hakee satunnaisen sanan tietokannasta ja palauttaa sen listana."""
    wordle_id = [random.randint(1, 1426)]
    cursor.execute("SELECT sana FROM wordle WHERE id = %s", wordle_id)
    rows = cursor.fetchall()

    for (word,) in rows:
        vastaus = list(word)

    debug_log(f"Haettu sana (listana): {vastaus}")
    return vastaus


def checker_wordle(vastaus, arvaus_list):
    """Tarkistaa arvatun sanan kirjaimet."""
    checker = []
    for i in range(5):
        if arvaus_list[i] == vastaus[i]:
            checker.append("green")
        elif arvaus_list[i] in vastaus:
            checker.append("yellow")
        else:
            checker.append("reset")
    return checker


def wordle_arvaus_def(arvaus):
    """Muuntaa annetun sanan listaksi."""
    return [kirjain for kirjain in arvaus]


def wordle_5letters():
    """Kysyy käyttäjältä syötteen ja varmistaa että se on 5-kirjaiminen."""
    while True:
        arvaus = input("")
        if len(arvaus) == 5 and isinstance(arvaus, str):
            break
    return arvaus.lower()


def wordle_tulostus(color_code, checker, arvaus, loop, vastaus):
    """Tulostaa yhden kierroksen rivin ja tarkistaa voiton."""
    # Huom: tulostus jätettiin koska osa pelin toimintaa
    win = False
    print(
        color_code[checker[0]] + f"[{arvaus[0]}]",
        color_code[checker[1]] + f"[{arvaus[1]}]",
        color_code[checker[2]] + f"[{arvaus[2]}]",
        color_code[checker[3]] + f"[{arvaus[3]}]",
        color_code[checker[4]] + f"[{arvaus[4]}]" + reset
    )

    if all(c == "green" for c in checker):
        loop += 6
        print("Onnittelut voitosta!")
        win = True
    else:
        loop += 1

    if loop >= 6 and not win:
        print(f"Hävisit pelin, oikea sana oli {vastaus}")

    return loop, win

    input("")

def wordle(loop, green, yellow, reset):
    """Pelin päälooppi."""
    vastaus_wordle = vastaus_wordle_def()
    ohjeet_wordle(green, yellow, reset)

    while loop < 6:
        arvaus = wordle_5letters()
        wordle_arvaus = wordle_arvaus_def(arvaus)
        wordle_checker = checker_wordle(vastaus_wordle, wordle_arvaus)
        loop, voitto_wordle = wordle_tulostus(wordle_color_code, wordle_checker, wordle_arvaus, loop, vastaus_wordle)

    return voitto_wordle

    "reset": "\033[0m"
}

yellow = "\033[33m"
reset = "\033[0m"
